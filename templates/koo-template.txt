class Koo < Formula
  desc "Deploy to Koo from your terminal"
  homepage "https://koo.io"
  version "{VERSION}"
  license "MIT"
{ON_MACOS}
  on_linux do
    if Hardware::CPU.arm?
      url "{URL}/koo-aarch64-unknown-linux-gnu.tar.gz",
          verified: "{VERIFIED}"
      sha256 "{SHA_AARCH64}"
    else
      url "{URL}/koo-x86_64-unknown-linux-gnu.tar.gz",
          verified: "{VERIFIED}"
      sha256 "{SHA_X86_64}"
    end
  end

  def install
    bin.install "koo"
  end

  test do
    assert_match "koo", shell_output("#{bin}/koo --help")
  end
end
